# LAN joiner: boot to the main menu, Multiplayer, Join Game, wait for the host to start.
mash 100 20 scene == 0
log STATE title screen
mash 100 30 scene != 0
wait scene == 1 || scene == 10
wait scene == 10 || nbuttons >= 8
if scene == 10
log first run: no save file, accepting 'create save data'
wait nbuttons >= 2
idle 10
mash 100 30 scene == 1
wait nbuttons >= 8
endif
wait onctx == 0
log STATE main menu
idle 30
# The main menu is a ring picked by stick/D-pad angle; Multiplayer (ctx 1) is up-left.
hold 30 0 -80 80
wait onctx == 1 120
idle 10
log STATE pointer on multiplayer
idle 20
press 100
wait scene == 10 || scene == 24 2400
if scene == 10
# Explicit mode choices; MSC_NET_MODE host/join shortcuts already enter the lobby.
log STATE multiplayer mode choice
wait nbuttons >= 3 900
idle 60
mash 4 15 onctx == 1 180
idle 10
press 100
wait scene == 1 900
wait scene == 10 900
wait nbuttons >= 3 900
idle 60
mash 8 15 onctx == 0 180
idle 10
press 100
log chose local nearby
endif
wait scene == 24 2400
log STATE lobby
# (mp18) wait until the host's game is listed: a fixed idle raced the host when the two consoles ran at different frame rates (LOCKED 60 against AUTO)
wait found >= 1 7200
idle 60
press 4
idle 30
press 100
log joining
wait peers >= 2 7200
log STATE joined
wait scene != 24 7200
log STATE launched
# Draft, captains, sidekicks, stadium: accept whatever is highlighted until the match is loaded.
mash 100 45 ingame == 1 12000
log STATE in match
wait gs == 5 6000
log STATE open play
idle 1000
log STATE leaving abruptly
exit
