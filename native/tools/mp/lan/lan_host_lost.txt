# LAN host whose own link is lost in the match (wave 5 NET: MSC_LAN_TEST_LINKDOWN / MSC_LAN_TEST_RESUME): the game's error popup must come
# at once, the match ends, the lobby follows (it finds nobody with a dead link) and the player backs out to the main menu. Based on lan_host_dc.txt.
# Scene ids: 1 main menu, 24 network lobby.
mash 100 20 scene == 0
log STATE title screen
mash 100 30 scene != 0
wait scene == 1 || scene == 10
wait scene == 10 || nbuttons >= 8
if scene == 10
log first run: no save file, accepting 'create save data'
wait nbuttons >= 2
idle 10
mash 100 30 scene == 1
wait nbuttons >= 8
endif
wait onctx == 0
log STATE main menu
idle 30
# The main menu is a ring picked by stick/D-pad angle; Multiplayer (ctx 1) is up-left.
hold 30 0 -80 80
wait onctx == 1 120
idle 10
log STATE pointer on multiplayer
idle 20
press 100
wait scene == 24 2400
log STATE lobby
idle 120
press 100
log created game
wait peers >= 2 7200
log STATE peer joined
idle 120
mash 100 60 scene != 24 3600
log STATE launched
# Draft, captains, sidekicks, stadium: accept whatever is highlighted until the match is loaded.
mash 100 45 ingame == 1 12000
log STATE in match
wait gs == 5 6000
log STATE open play
# This console's link is declared lost in the match: the game must notice at its next send and leave the match.
wait scene == 10 || ovl == 10 7200
log STATE error popup
idle 120
mash 100 90 ingame == 0 3600
log STATE match ended after disconnect
wait scene == 1 || scene == 24 3600
if scene == 24
log STATE back in lobby
idle 300
press 200
endif
wait scene == 1 2400
log STATE main menu again
wait nbuttons >= 8 600
idle 120
exit
