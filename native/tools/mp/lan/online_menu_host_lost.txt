# Online lobby with the link cut: see online_menu_test.py (case lost). Host side.
# Online host through the game's own menus: boot to the main menu, MULTIPLAYER, HOST A ROOM (name by the test keyboard, account on the service),
# wait for the partner (status popup), Create Game, start, follow the game.
# start, then follow whatever the game shows. Scene ids: 1 main menu, 24 network lobby.
mash 100 20 scene == 0
log STATE title screen
mash 100 30 scene != 0
wait scene == 1 || scene == 10
wait scene == 10 || nbuttons >= 8
if scene == 10
log first run: no save file, accepting 'create save data'
wait nbuttons >= 2
idle 10
mash 100 30 scene == 1
wait nbuttons >= 8
endif
wait onctx == 0
log STATE main menu
idle 30
# The main menu is a ring picked by stick/D-pad angle; Multiplayer (ctx 1) is up-left.
hold 30 0 -80 80
wait onctx == 1 120
idle 10
log STATE pointer on multiplayer
idle 20
press 100
wait scene == 10 1200
log STATE multiplayer mode choice
wait nbuttons >= 3 900
idle 60
mash 8 15 onctx == 0 180
idle 10
press 100
log chose online
wait scene == 1 900
wait scene == 10 1200
log STATE choice popup
wait nbuttons >= 3 900
idle 20
press 100
log chose host
# the name (test keyboard), the account, the room and the wait for the partner are popups of the game; the lobby follows once the link is up
wait scene == 24 7200
log STATE lobby
idle 120
press 100
log created game
wait peers >= 2 7200
log STATE peer joined
# The test cuts the route ("down 0") 2 s after the line above. The game's own keep-alive (MSC_LAN_KEEPALIVE_MS) ends the link: its "connection lost"
# popup (OK), then the lobby offers RECONNECT / LEAVE.
wait scene == 10 5400
log STATE connection lost popup
wait nbuttons >= 1 600
idle 45
press 100
log STATE connection lost answered
wait scene == 10 && nbuttons >= 2 5400
log STATE reconnect offered
idle 60
press 100
log chose reconnect
# the status popup of the rebuild is up for a few seconds: let it come and go, then the lobby (scene 24) is on top again
idle 150
wait scene == 24 5400
log STATE lobby restored
idle 150
press 100
log created game again
wait peers >= 2 5400
log STATE peer joined again
# Match the joiner's final hold, so normal teardown does not become a second link-loss case.
idle 300
exit
